  `username` VARCHAR(255) NOT NULL UNIQUE,
  `password_hash` VARCHAR(255) NOT NULL,

);

-- This table caches geolocation data for IP addresses.
CREATE TABLE `ip_geolocation_cache` (
  `ip_address` VARCHAR(45) PRIMARY KEY,
  `country` VARCHAR(255) NULL,
  `region` VARCHAR(255) NULL,
  `city` VARCHAR(255) NULL,
  `lat` DECIMAL(10, 8) NULL,
  `lon` DECIMAL(11, 8) NULL,
  `isp` VARCHAR(255) NULL,
  `is_proxy` TINYINT(1) NULL,
  `is_hosting` TINYINT(1) NULL,
  `last_updated` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);


-- This table stores the processed sign-in logs from Microsoft Entra.
CREATE TABLE `login_logs` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `entra_log_id` VARCHAR(255) NOT NULL UNIQUE,
  `user_id` VARCHAR(255) NOT NULL,
  `user_principal_name` VARCHAR(255) NOT NULL,
  `ip_address` VARCHAR(45) NOT NULL,
  `login_time` DATETIME NOT NULL,
  `status` TEXT NULL,
  `country` VARCHAR(255) NULL,
  `region` VARCHAR(255) NULL,
  `city` VARCHAR(255) NULL,
  `lat` DECIMAL(10, 8) NULL,
  `lon` DECIMAL(11, 8) NULL,
  `is_impossible_travel` BOOLEAN NOT NULL DEFAULT FALSE,
  `is_region_change` BOOLEAN NOT NULL DEFAULT FALSE,
  `travel_speed_kph` DECIMAL(10, 2) NULL,
  `previous_log_id` INT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  -- Context captured for risk scoring (see src/RiskScorer.php)
  `error_code` INT NULL,
  `app_display_name` VARCHAR(255) NULL,
  `resource_display_name` VARCHAR(255) NULL,
  `client_app_used` VARCHAR(100) NULL,
  `device_id` VARCHAR(100) NULL,
  `device_display_name` VARCHAR(255) NULL,
  `device_os` VARCHAR(100) NULL,
  `device_browser` VARCHAR(255) NULL,
  `is_compliant` TINYINT(1) NULL,
  `is_managed` TINYINT(1) NULL,
  `is_interactive` TINYINT(1) NULL,
  `conditional_access_status` VARCHAR(50) NULL,
  `risk_level_during_signin` VARCHAR(30) NULL,
  `risk_state` VARCHAR(50) NULL,
  `isp` VARCHAR(255) NULL,
  `is_proxy` TINYINT(1) NULL,
  `is_hosting` TINYINT(1) NULL,
  INDEX `user_id_idx` (`user_id`),
  INDEX `login_time_idx` (`login_time`),
  INDEX `user_time_idx` (`user_id`, `login_time`),
  INDEX `ip_time_idx` (`ip_address`, `login_time`)
);


-- This table logs every time a consolidated email alert is sent.
CREATE TABLE `email_alerts` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `alert_log_id` INT NOT NULL,
  `compared_log_id` INT NOT NULL,
  `alert_type` VARCHAR(50) NOT NULL,
  `sent_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  INDEX `alert_log_id_idx` (`alert_log_id`),
  INDEX `compared_log_id_idx` (`compared_log_id`)
);


-- This table stores whitelisted IP addresses that should not trigger email alerts.
CREATE TABLE `ip_whitelist` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `ip_address` VARCHAR(45) NOT NULL UNIQUE,
  `note` TEXT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- This table stores a snapshot of a user's currently registered authentication devices.
CREATE TABLE `user_auth_devices` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` VARCHAR(255) NOT NULL,
  `device_id` VARCHAR(255) NOT NULL,
  `display_name` VARCHAR(255) NULL,
  `device_type` VARCHAR(255) NULL,
  `last_seen` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY `user_device_unique` (`user_id`, `device_id`)
);


-- This table logs every time a device is added or removed for a user.
CREATE TABLE `auth_device_changes` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` VARCHAR(255) NOT NULL,
  `user_principal_name` VARCHAR(255) NOT NULL,
  `device_display_name` VARCHAR(255) NULL,
  `change_type` VARCHAR(50) NOT NULL,
  `change_time` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  INDEX `user_id_idx` (`user_id`),
  INDEX `change_time_idx` (`change_time`)
);

-- Per-user rolling profile of "normal": countries, ISPs, apps, devices, client apps.
-- days_seen counts distinct calendar days a value was observed; a value only
-- counts as "established" once days_seen reaches BASELINE_CONFIRM_DAYS.
CREATE TABLE `user_baselines` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` VARCHAR(255) NOT NULL,
  `dimension` VARCHAR(20) NOT NULL,
  `value` VARCHAR(255) NOT NULL,
  `seen_count` INT NOT NULL DEFAULT 1,
  `days_seen` INT NOT NULL DEFAULT 1,
  `first_seen` DATETIME NOT NULL,
  `last_seen` DATETIME NOT NULL,
  UNIQUE KEY `user_dim_value_unique` (`user_id`, `dimension`, `value`),
  INDEX `last_seen_idx` (`last_seen`)
);

-- One row per fired risk signal. dedupe_key stops the same signal being
-- recorded repeatedly for the same underlying activity.
CREATE TABLE `risk_events` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` VARCHAR(255) NOT NULL,
  `user_principal_name` VARCHAR(255) NOT NULL,
  `signal_code` VARCHAR(50) NOT NULL,
  `weight` SMALLINT NOT NULL,
  `login_log_id` INT NULL,
  `device_change_id` INT NULL,
  `ip_address` VARCHAR(45) NULL,
  `description` VARCHAR(500) NOT NULL,
  `details` TEXT NULL,
  `occurred_at` DATETIME NOT NULL,
  `dedupe_key` VARCHAR(190) NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY `event_dedupe_unique` (`user_id`, `signal_code`, `dedupe_key`),
  INDEX `user_time_idx` (`user_id`, `occurred_at`),
  INDEX `occurred_idx` (`occurred_at`)
);

-- Current rolled-up risk per user (rebuilt from risk_events on every run).
CREATE TABLE `user_risk_scores` (
  `user_id` VARCHAR(255) PRIMARY KEY,
  `user_principal_name` VARCHAR(255) NOT NULL,
  `score` SMALLINT NOT NULL,
  `tier` VARCHAR(10) NOT NULL,
  `signals` TEXT NULL,
  `event_count` INT NOT NULL DEFAULT 0,
  `last_event_at` DATETIME NULL,
  `updated_at` DATETIME NOT NULL,
  INDEX `score_idx` (`score`)
);
